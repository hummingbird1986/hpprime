// Pixel width of s in font f, measured on the scratch GROB G9 (sized in zMAIN).
zTW(s, f)
BEGIN
  RETURN TEXTOUT_P(s, G9, 0, 0, f);
END;

zCT(s, cx, y, f, c)
BEGIN
  TEXTOUT_P(s, G0, cx - IP(zTW(s, f)/2), y, f, c);
END;

zRT(s, rx, y, f, c)
BEGIN
  TEXTOUT_P(s, G0, rx - zTW(s, f), y, f, c);
END;

// A number as text in Standard format, 6 significant digits, whatever the
// Home number format is. Strings pass through ("n/a", "∞").
zN(v)
BEGIN
  IF TYPE(v) == 2 THEN RETURN v; END;
  IF v == IP(v) AND ABS(v) < 10^12 THEN RETURN STRING(v, 1); END;
  RETURN STRING(ROUND(v, -6), 1);
END;

// The significance text for the current tail: "α = 0.05", or for Two
// "α/2 = 0.025" (each tail; the doubled p is still compared with α).
zAT()
BEGIN
  IF PCt == 3 THEN RETURN "α/2 = " + zN(PCa/2); END;
  RETURN "α = " + zN(PCa);
END;

// b^e with 0^0 = 1 (geometric / negative binomial with p = 1).
zPW(b, ex)
BEGIN
  IF ex == 0 THEN RETURN 1; END;
  RETURN b^ex;
END;

// The EEX mark (U+1D07), built with CHAR (see Base16C).
zEE()
BEGIN
  RETURN CHAR(7431);
END;

// 1 if string s contains character c (INSTRING fails on an empty string).
zHAS(s, c)
BEGIN
  LOCAL j;
  FOR j FROM 1 TO DIM(s) DO
    IF MID(s, j, 1) == c THEN RETURN 1; END;
  END;
  RETURN 0;
END;

zDS()
BEGIN
  RETURN PCd >= 8;
END;

// Name of the variable for the quantile keys: z, t, χ², F, else x.
zQS()
BEGIN
  CASE
    IF PCd == 2 THEN RETURN "z"; END;
    IF PCd == 3 THEN RETURN "t"; END;
    IF PCd == 4 THEN RETURN "χ²"; END;
    IF PCd == 5 THEN RETURN "F"; END;
  END;
  RETURN "x";
END;

// 1 if the CI key applies: Normal, Std Normal, t, Binomial, Bernoulli.
zCIOK()
BEGIN
  RETURN PCd <= 3 OR PCd == 8 OR PCd == 9;
END;

// 1 if calc-menu code c is shown for the current distribution
// (CI: see zCIOK; Test: Normal, Std Normal, t, χ², F).
zKOK(c)
BEGIN
  IF c == 15 THEN RETURN zCIOK(); END;
  IF c == 16 THEN RETURN PCd <= 5; END;
  RETURN 1;
END;

// Support {lo, hi} of a discrete distribution (hi = 10^9: unbounded).
zSUP()
BEGIN
  LOCAL g := PCp(PCd);
  CASE
    IF PCd == 8  THEN RETURN {0, g(1)}; END;
    IF PCd == 9  THEN RETURN {0, 1}; END;
    IF PCd == 10 THEN RETURN {1, 10^9}; END;
    IF PCd == 11 THEN RETURN {g(1), 10^9}; END;
    IF PCd == 12 THEN RETURN {0, 10^9}; END;
    IF PCd == 13 THEN RETURN {MAX(0, g(3)-(g(1)-g(2))), MIN(g(3), g(2))}; END;
    IF PCd == 14 THEN RETURN {g(3), g(1)-g(2)+g(3)}; END;
    DEFAULT RETURN {-10^99, 10^99};
  END;
END;

// Density f(x) (continuous) or P(X = x) (discrete).
zPDF(x)
BEGIN
  LOCAL g := PCp(PCd);
  LOCAL s;
  CASE
    IF PCd == 1 THEN RETURN NORMALD(g(1), g(2), x); END;
    IF PCd == 2 THEN RETURN NORMALD(0, 1, x); END;
    IF PCd == 3 THEN RETURN STUDENT(g(1), x); END;
    IF PCd == 4 THEN
      IF x <= 0 THEN RETURN 0; END;
      RETURN CHISQUARE(g(1), x);
    END;
    IF PCd == 5 THEN
      IF x <= 0 THEN RETURN 0; END;
      RETURN FISHER(g(1), g(2), x);
    END;
    IF PCd == 6 THEN
      IF x < 0 THEN RETURN 0; END;
      RETURN g(1)*EXP(-g(1)*x);
    END;
    IF PCd == 7 THEN
      IF x < g(1) OR x > g(2) THEN RETURN 0; END;
      RETURN 1/(g(2)-g(1));
    END;
  END;
  IF x <> FLOOR(x) THEN RETURN 0; END;
  s := zSUP();
  IF x < s(1) OR x > s(2) THEN RETURN 0; END;
  CASE
    IF PCd == 8  THEN RETURN BINOMIAL(g(1), g(2), x); END;
    IF PCd == 9  THEN RETURN IFTE(x == 1, g(1), 1-g(1)); END;
    IF PCd == 10 THEN RETURN g(1)*zPW(1-g(1), x-1); END;
    IF PCd == 11 THEN RETURN COMB(x-1, g(1)-1)*g(2)^g(1)*zPW(1-g(2), x-g(1)); END;
    IF PCd == 12 THEN RETURN POISSON(g(1), x); END;
    IF PCd == 13 THEN RETURN COMB(g(2), x)*COMB(g(1)-g(2), g(3)-x)/COMB(g(1), g(3)); END;
    DEFAULT RETURN COMB(x-1, g(3)-1)*COMB(g(1)-x, g(2)-g(3))/COMB(g(1), g(2));
  END;
END;

// P(X <= x).
zCDF(x)
BEGIN
  LOCAL g := PCp(PCd);
  LOCAL s, k, j, c;
  CASE
    IF PCd == 1 THEN RETURN NORMALD_CDF(g(1), g(2), x); END;
    IF PCd == 2 THEN RETURN NORMALD_CDF(0, 1, x); END;
    IF PCd == 3 THEN RETURN STUDENT_CDF(g(1), x); END;
    IF PCd == 4 THEN
      IF x <= 0 THEN RETURN 0; END;
      RETURN CHISQUARE_CDF(g(1), x);
    END;
    IF PCd == 5 THEN
      IF x <= 0 THEN RETURN 0; END;
      RETURN FISHER_CDF(g(1), g(2), x);
    END;
    IF PCd == 6 THEN
      IF x <= 0 THEN RETURN 0; END;
      RETURN 1-EXP(-g(1)*x);
    END;
    IF PCd == 7 THEN
      IF x <= g(1) THEN RETURN 0; END;
      IF x >= g(2) THEN RETURN 1; END;
      RETURN (x-g(1))/(g(2)-g(1));
    END;
  END;
  k := FLOOR(x);
  s := zSUP();
  IF k < s(1) THEN RETURN 0; END;
  IF k >= s(2) THEN RETURN 1; END;
  CASE
    IF PCd == 8  THEN RETURN BINOMIAL_CDF(g(1), g(2), k); END;
    IF PCd == 9  THEN RETURN 1-g(1); END;
    IF PCd == 10 THEN RETURN 1-zPW(1-g(1), k); END;
    IF PCd == 12 THEN RETURN POISSON_CDF(g(1), k); END;
  END;
  c := 0;
  FOR j FROM s(1) TO k DO c := c + zPDF(j); END;
  RETURN MIN(1, c);
END;

// P(X >= x). Normal and t use the mirror image to keep small tails accurate.
zSF(x)
BEGIN
  LOCAL g := PCp(PCd);
  CASE
    IF PCd == 1 THEN RETURN NORMALD_CDF(g(1), g(2), 2*g(1)-x); END;
    IF PCd == 2 THEN RETURN NORMALD_CDF(0, 1, -x); END;
    IF PCd == 3 THEN RETURN STUDENT_CDF(g(1), -x); END;
    IF PCd == 6 THEN
      IF x <= 0 THEN RETURN 1; END;
      RETURN EXP(-g(1)*x);
    END;
  END;
  IF zDS() THEN RETURN MAX(0, 1-zCDF(CEILING(x)-1)); END;
  RETURN MAX(0, 1-zCDF(x));
END;

// Lower quantile: continuous x with P(X<=x) = q; discrete smallest x with
// P(X<=x) >= q.
zICDF(q)
BEGIN
  LOCAL g := PCp(PCd);
  LOCAL s, lo, hi, k, c, cap;
  CASE
    IF PCd == 1 THEN RETURN NORMALD_ICDF(g(1), g(2), q); END;
    IF PCd == 2 THEN RETURN NORMALD_ICDF(0, 1, q); END;
    IF PCd == 3 THEN RETURN STUDENT_ICDF(g(1), q); END;
    IF PCd == 4 THEN RETURN CHISQUARE_ICDF(g(1), q); END;
    IF PCd == 5 THEN RETURN FISHER_ICDF(g(1), g(2), q); END;
    IF PCd == 6 THEN RETURN -LN(1-q)/g(1); END;
    IF PCd == 7 THEN RETURN g(1)+q*(g(2)-g(1)); END;
  END;
  s := zSUP(); lo := s(1); hi := s(2);
  IF q <= 0 THEN RETURN lo; END;
  IF q >= 1 AND hi < 10^9 THEN RETURN hi; END;
  q := MIN(q, 1-10^-12);
  // A starting guess, then step to the exact boundary.
  k := -1;
  IFERR
    IF PCd == 8 THEN k := BINOMIAL_ICDF(g(1), g(2), q); END;
    IF PCd == 12 THEN k := POISSON_ICDF(g(1), q); END;
    IF PCd == 10 THEN
      IF g(1) >= 1 THEN k := 1; ELSE k := CEILING(LN(1-q)/LN(1-g(1))); END;
    END;
  THEN
    k := -1;
  END;
  IF k >= 0 THEN
    k := MAX(lo, MIN(hi, k));
    WHILE k > lo AND zCDF(k-1) >= q-10^-12 DO k := k-1; END;
    WHILE k < hi AND zCDF(k) < q-10^-12 DO k := k+1; END;
    RETURN k;
  END;
  k := lo; c := zPDF(lo);
  cap := MIN(hi, lo+200000);
  WHILE c < q-10^-12 AND k < cap DO
    k := k+1;
    c := c + zPDF(k);
  END;
  RETURN k;
END;

// Discrete: largest x with P(X<=x) <= q (lo-1 if there is none).
zLQ(q)
BEGIN
  LOCAL k;
  k := zICDF(q);
  IF zCDF(k) > q+10^-12 THEN k := k-1; END;
  RETURN k;
END;

// Upper quantile: continuous x with P(X>=x) = q; discrete smallest x with
// P(X>=x) <= q.
zISF(q)
BEGIN
  LOCAL g := PCp(PCd);
  LOCAL s;
  CASE
    IF PCd == 1 THEN RETURN 2*g(1)-NORMALD_ICDF(g(1), g(2), q); END;
    IF PCd == 2 THEN RETURN -NORMALD_ICDF(0, 1, q); END;
    IF PCd == 3 THEN RETURN -STUDENT_ICDF(g(1), q); END;
    IF PCd == 6 THEN RETURN -LN(q)/g(1); END;
    IF PCd == 7 THEN RETURN g(2)-q*(g(2)-g(1)); END;
  END;
  IF NOT zDS() THEN RETURN zICDF(1-q); END;
  s := zSUP();
  IF q >= 1 THEN RETURN s(1); END;
  RETURN zICDF(1-q)+1;
END;

// {E(X), Var(X)}; a string where it is undefined.
zMV()
BEGIN
  LOCAL g := PCp(PCd);
  LOCAL m, v;
  m := "n/a"; v := "n/a";
  CASE
    IF PCd == 1 THEN m := g(1); v := g(2)^2; END;
    IF PCd == 2 THEN m := 0; v := 1; END;
    IF PCd == 3 THEN
      IF g(1) > 1 THEN m := 0; v := "∞"; END;
      IF g(1) > 2 THEN v := g(1)/(g(1)-2); END;
    END;
    IF PCd == 4 THEN m := g(1); v := 2*g(1); END;
    IF PCd == 5 THEN
      IF g(2) > 2 THEN m := g(2)/(g(2)-2); v := "∞"; END;
      IF g(2) > 4 THEN v := 2*g(2)^2*(g(1)+g(2)-2)/(g(1)*(g(2)-2)^2*(g(2)-4)); END;
    END;
    IF PCd == 6 THEN m := 1/g(1); v := 1/g(1)^2; END;
    IF PCd == 7 THEN m := (g(1)+g(2))/2; v := (g(2)-g(1))^2/12; END;
    IF PCd == 8 THEN m := g(1)*g(2); v := g(1)*g(2)*(1-g(2)); END;
    IF PCd == 9 THEN m := g(1); v := g(1)*(1-g(1)); END;
    IF PCd == 10 THEN m := 1/g(1); v := (1-g(1))/g(1)^2; END;
    IF PCd == 11 THEN m := g(1)/g(2); v := g(1)*(1-g(2))/g(2)^2; END;
    IF PCd == 12 THEN m := g(1); v := g(1); END;
    IF PCd == 13 THEN
      m := g(3)*g(2)/g(1);
      v := 0;
      IF g(1) > 1 THEN v := g(3)*(g(2)/g(1))*(1-g(2)/g(1))*(g(1)-g(3))/(g(1)-1); END;
    END;
    IF PCd == 14 THEN
      m := g(3)*(g(1)+1)/(g(2)+1);
      v := g(3)*(g(1)-g(2))*(g(1)+1)*(g(2)+1-g(3))/((g(2)+1)^2*(g(2)+2));
    END;
  END;
  RETURN {m, v};
END;

// "" if parameter list g is valid for the current distribution, else why not.
zCHK(g)
BEGIN
  LOCAL d := PCd;
  CASE
    IF d == 1 AND g(2) <= 0 THEN RETURN "σ must be > 0"; END;
    IF (d == 3 OR d == 4) AND g(1) <= 0 THEN RETURN "df must be > 0"; END;
    IF d == 5 AND (g(1) <= 0 OR g(2) <= 0) THEN RETURN "df must be > 0"; END;
    IF (d == 6 OR d == 12) AND g(1) <= 0 THEN RETURN "λ must be > 0"; END;
    IF d == 7 AND g(2) <= g(1) THEN RETURN "Need b > a"; END;
    IF d == 8 AND (g(1) < 0 OR g(1) <> IP(g(1))) THEN RETURN "n: integer >= 0"; END;
    IF d == 8 AND (g(2) < 0 OR g(2) > 1) THEN RETURN "Need 0 <= p <= 1"; END;
    IF d == 9 AND (g(1) < 0 OR g(1) > 1) THEN RETURN "Need 0 <= p <= 1"; END;
    IF d == 10 AND (g(1) <= 0 OR g(1) > 1) THEN RETURN "Need 0 < p <= 1"; END;
    IF d == 11 AND (g(1) < 1 OR g(1) <> IP(g(1))) THEN RETURN "k: integer >= 1"; END;
    IF d == 11 AND (g(2) <= 0 OR g(2) > 1) THEN RETURN "Need 0 < p <= 1"; END;
  END;
  IF d == 13 OR d == 14 THEN
    IF g(1) <> IP(g(1)) OR g(2) <> IP(g(2)) OR g(3) <> IP(g(3)) THEN RETURN "N, K, n/r: integers"; END;
    IF g(1) < 1 OR g(2) < 0 OR g(2) > g(1) THEN RETURN "Need 0 <= K <= N"; END;
    IF d == 13 AND (g(3) < 0 OR g(3) > g(1)) THEN RETURN "Need 0 <= n <= N"; END;
    IF d == 14 AND (g(3) < 1 OR g(3) > g(2)) THEN RETURN "Need 1 <= r <= K"; END;
  END;
  RETURN "";
END;

// p for the current tail at x (Between: P(a<=X<=b) with a, b = x, y).
zPV()
BEGIN
  LOCAL a, b;
  CASE
    IF PCt == 1 THEN RETURN zCDF(PCx); END;
    IF PCt == 2 THEN RETURN zSF(PCx); END;
    IF PCt == 3 THEN RETURN MIN(1, 2*MIN(zCDF(PCx), zSF(PCx))); END;
  END;
  a := MIN(PCx, PCy); b := MAX(PCx, PCy);
  IF zDS() THEN RETURN MAX(0, zCDF(FLOOR(b)) - zCDF(CEILING(a)-1)); END;
  RETURN MAX(0, zCDF(b) - zCDF(a));
END;

// Shading bounds {b1, b2}: Left X<=b1, Right X>=b1, Two X<=b1 or X>=b2
// (the far tail holds the same probability as the near one), Between b1..b2.
zTB()
BEGIN
  LOCAL x, pl, pr, lo, hi;
  x := PCx;
  IF PCt <= 2 THEN RETURN {x, x}; END;
  IF PCt == 4 THEN RETURN {MIN(x, PCy), MAX(x, PCy)}; END;
  pl := zCDF(x); pr := zSF(x);
  IF pl <= pr THEN
    lo := x; hi := 10^99;
    IF pl > 0 THEN
      IFERR hi := zISF(pl); THEN hi := 10^99; END;
    END;
  ELSE
    hi := x; lo := -10^99;
    IF pr > 0 THEN
      IFERR
        IF zDS() THEN lo := zLQ(pr); ELSE lo := zICDF(pr); END;
      THEN
        lo := -10^99;
      END;
    END;
  END;
  RETURN {lo, hi};
END;

// 1 if value v lies in the shaded region.
zIN(v, tb)
BEGIN
  CASE
    IF PCt == 1 THEN RETURN v <= tb(1); END;
    IF PCt == 2 THEN RETURN v >= tb(1); END;
    IF PCt == 3 THEN RETURN v <= tb(1) OR v >= tb(2); END;
    DEFAULT RETURN v >= tb(1) AND v <= tb(2);
  END;
END;

// Critical values at alpha for the current tail: {{value, side}}, side -1 for
// a lower rejection region (X <= c), +1 for an upper one (X >= c).
zCRIT()
BEGIN
  LOCAL a, r;
  a := PCa; r := {};
  IFERR
    IF PCt == 1 THEN r := {{IFTE(zDS(), zLQ(a), zICDF(a)), -1}}; END;
    IF PCt == 2 THEN r := {{zISF(a), 1}}; END;
    IF PCt == 3 THEN
      IF zDS() THEN r := {{zLQ(a/2), -1}}; ELSE r := {{zICDF(a/2), -1}}; END;
      r(2) := {zISF(a/2), 1};
    END;
  THEN
    r := {};
  END;
  RETURN r;
END;

// A starting x for a new distribution: the critical value at alpha for the
// current tail, so the plot opens on a shaded area of exactly alpha.
zDEFX()
BEGIN
  IFERR
    CASE
      IF PCt == 1 THEN PCx := zICDF(PCa); END;
      IF PCt == 2 THEN PCx := zISF(PCa); END;
      IF PCt == 3 THEN PCx := zISF(PCa/2); END;
      DEFAULT PCx := zICDF(0.25); PCy := zISF(0.25);
    END;
  THEN
    PCx := 0; PCy := 1;
  END;
END;

// Name, title "X ~ N(0, 1)" and parameter line of the current distribution.
zTTL()
BEGIN
  LOCAL g := PCp(PCd);
  LOCAL s, k;
  IF PCd == 2 THEN RETURN "Z ~ N(0, 1)"; END;
  s := "";
  FOR k FROM 1 TO SIZE(g) DO
    IF k > 1 THEN s := s + ", "; END;
    s := s + zN(g(k));
  END;
  RETURN "X ~ " + zSY(PCd) + "(" + s + ")";
END;

zPLN()
BEGIN
  LOCAL g := PCp(PCd);
  LOCAL nm := zPNL(PCd);
  LOCAL s, k;
  IF SIZE(g) == 0 THEN RETURN "μ = 0   σ = 1"; END;
  s := "";
  FOR k FROM 1 TO SIZE(g) DO
    IF k > 1 THEN s := s + "    "; END;
    s := s + nm(k) + " " + zN(g(k));
  END;
  RETURN s;
END;

// A round tick step for the range u0..u1 (at least 1 for discrete).
zNS(u0, u1)
BEGIN
  LOCAL st, mg, n;
  st := (u1-u0)/5;
  IF st <= 0 THEN RETURN 1; END;
  mg := 10^FLOOR(LOG(st));
  n := st/mg;
  st := IFTE(n < 1.5, 1, IFTE(n < 3, 2, IFTE(n < 7, 5, 10)))*mg;
  IF zDS() THEN st := MAX(1, ROUND(st, 0)); END;
  RETURN st;
END;

// Plot range {u0, u1} in X units. Discrete: bars are centred on integers, so
// the range runs from lo-0.5 to hi+0.5. Widened to show x (and y in Between)
// unless it is far off scale.
zRNG()
BEGIN
  LOCAL g := PCp(PCd);
  LOCAL u0, u1, w, q, s, lo, hi, v;
  LOCAL vs, k;
  vs := {PCx};
  IF PCt == 4 THEN vs(2) := PCy; END;
  IF zDS() THEN
    s := zSUP(); lo := s(1); hi := s(2);
    IF hi - lo > 40 THEN
      lo := MAX(lo, zICDF(0.0005));
      hi := MIN(hi, zICDF(0.9995));
    END;
    w := hi - lo + 1;
    FOR k FROM 1 TO SIZE(vs) DO
      v := vs(k);
      IF v < lo AND v > lo-2*w THEN lo := MAX(s(1), FLOOR(v)); END;
      IF v > hi AND v < hi+2*w THEN hi := MIN(s(2), CEILING(v)); END;
    END;
    IF hi - lo > 300 THEN hi := lo + 300; END;
    RETURN {lo-0.5, hi+0.5};
  END;
  CASE
    IF PCd == 1 THEN u0 := g(1)-4*g(2); u1 := g(1)+4*g(2); END;
    IF PCd == 2 THEN u0 := -4; u1 := 4; END;
    IF PCd == 3 THEN
      q := MIN(8, MAX(4, STUDENT_ICDF(g(1), 0.995)));
      u0 := -q; u1 := q;
    END;
    IF PCd == 4 THEN u0 := 0; u1 := CHISQUARE_ICDF(g(1), 0.999); END;
    IF PCd == 5 THEN u0 := 0; u1 := MIN(12, MAX(4, FISHER_ICDF(g(1), g(2), 0.99))); END;
    IF PCd == 6 THEN u0 := 0; u1 := -LN(0.001)/g(1); END;
    DEFAULT
      w := g(2)-g(1);
      u0 := g(1)-w/4; u1 := g(2)+w/4;
  END;
  w := u1 - u0;
  FOR k FROM 1 TO SIZE(vs) DO
    v := vs(k);
    IF v < u0 AND v > u0-2*w THEN u0 := v - 0.05*w; END;
    IF v > u1 AND v < u1+2*w THEN u1 := v + 0.05*w; END;
  END;
  IF PCd >= 4 AND PCd <= 6 THEN u0 := MAX(0, u0); END;
  RETURN {u0, u1};
END;

zSET()
BEGIN
  LOCAL g := PCp(PCd);
  LOCAL nm := zPNL(PCd);
  LOCAL hp := zPHL(PCd);
  LOCAL q1, q2, q3, ok, er, t;
  IF SIZE(g) == 0 THEN zMsg := "No parameters"; RETURN; END;
  q1 := g(1);
  IF SIZE(g) >= 2 THEN q2 := g(2); END;
  IF SIZE(g) >= 3 THEN q3 := g(3); END;
  CASE
    IF SIZE(g) == 1 THEN ok := INPUT({q1}, zDN(PCd), nm, hp); g := {q1}; END;
    IF SIZE(g) == 2 THEN ok := INPUT({q1, q2}, zDN(PCd), nm, hp); g := {q1, q2}; END;
    DEFAULT ok := INPUT({q1, q2, q3}, zDN(PCd), nm, hp); g := {q1, q2, q3};
  END;
  IF NOT ok THEN RETURN; END;
  er := zCHK(g);
  IF DIM(er) > 0 THEN zMsg := er; RETURN; END;
  t := PCp;
  t(PCd) := g;
  PCp := t;
  zADD("Set", zTTL());
END;

// A probability argument for the quantile keys: 0 < p < 1.
zPRB(t)
BEGIN
  LOCAL r;
  r := zGV();
  IF r(1) == -1 THEN RETURN {0, 0}; END;
  IF r(1) == 0 THEN
    RETURN {1, IFTE(zLP > 0 AND zLP < 1, zLP, PCa)};
  END;
  IF r(2) <= 0 OR r(2) >= 1 THEN zMsg := "Need 0 < p < 1"; RETURN {0, 0}; END;
  RETURN r;
END;

// Confidence interval at confidence 1-α (always two-sided, α/2 per side) and
// its margin of error. Normal / Std Normal: z-interval for μ (σ known);
// Student t: t-interval for μ (df = n-1); Binomial / Bernoulli: z-interval
// for a proportion. A typed number is the mean (or proportion).
zCI()
BEGIN
  LOCAL g, m, s, n, q, c, r, ok;
  LOCAL me, cf, st, ct;
  g := PCp(PCd);
  cf := zN(100*(1-PCa)) + "%";
  r := zGV();
  IF r(1) == -1 THEN RETURN; END;
  IF PCd <= 3 THEN
    m := zCIm; s := zCIs; n := zCIn;
    IF PCd == 1 THEN s := g(2); END;
    IF r(1) == 1 THEN m := r(2); END;
    IF PCd == 3 THEN
      ok := INPUT({m, s, n}, cf + " CI for μ (t)", {"Mean =", "s =", "n ="},
                  {"Sample mean", "Sample standard deviation s", "Sample size n ≥ 2"});
    ELSE
      ok := INPUT({m, s, n}, cf + " CI for μ (z)", {"Mean =", "σ =", "n ="},
                  {"Sample mean", "Population standard deviation σ", "Sample size n"});
    END;
    IF NOT ok THEN RETURN; END;
    IF s <= 0 THEN zMsg := "CI: need SD > 0"; RETURN; END;
    IF n <> IP(n) OR n < IFTE(PCd == 3, 2, 1) THEN zMsg := "CI: need a whole n ≥ " + IFTE(PCd == 3, "2", "1"); RETURN; END;
    zCIm := m; zCIs := s; zCIn := n;
    IF PCd == 3 THEN
      c := STUDENT_ICDF(n-1, 1-PCa/2); st := "t*(" + zN(n-1) + ")";
    ELSE
      c := NORMALD_ICDF(0, 1, 1-PCa/2); st := "z*";
    END;
    me := c*s/SQRT(n); ct := m;
  ELSE
    q := zCIp; n := zCIn;
    IF r(1) == 1 THEN q := r(2); END;
    ok := INPUT({q, n}, cf + " CI for p", {"p =", "n ="},
                {"Sample proportion (x/n is fine)", "Sample size n"});
    IF NOT ok THEN RETURN; END;
    IF q < 0 OR q > 1 THEN zMsg := "CI: need 0 ≤ p ≤ 1"; RETURN; END;
    IF n <> IP(n) OR n < 1 THEN zMsg := "CI: need a whole n ≥ 1"; RETURN; END;
    zCIp := q; zCIn := n;
    c := NORMALD_ICDF(0, 1, 1-PCa/2); st := "z*";
    me := c*SQRT(q*(1-q)/n); ct := q;
  END;
  zADD("MoE  " + st + " = " + zN(c), zN(me));
  zADD(cf + " CI", "[" + zN(ct-me) + ", " + zN(ct+me) + "]");
END;

// Remember the inputs v of test k.
zSV(k, v)
BEGIN
  LOCAL w := zTI;
  w(k) := v;
  zTI := w;
END;

// Common end of every test: switch to distribution d with parameters g
// (PCx already holds the statistic), Between becomes Two, then tape rows:
// header hd / hv, the statistic sn, p for the tail, and the 1-α CI for cp.
zTFIN(hd, hv, d, g, sn, lo, hi, cp)
BEGIN
  LOCAL w;
  w := PCp; w(d) := g; PCp := w;
  PCd := d;
  IF PCt == 4 THEN PCt := 3; END;
  zADD(hd, hv);
  zADD(sn + " statistic", zN(PCx));
  zADP(zPLB(), zPV());
  zADD(zN(100*(1-PCa)) + "% CI " + cp, "[" + zN(lo) + ", " + zN(hi) + "]");
END;

// z* for a two-sided 1-α interval.
zZS()
BEGIN
  RETURN NORMALD_ICDF(0, 1, 1-PCa/2);
END;

// 1-Z: one-sample z test for μ (σ known). r: typed value (the mean).
zZ1(r)
BEGIN
  LOCAL v, m, s, n, u0, se, c;
  v := zTI(1); m := v(1); s := v(2); n := v(3); u0 := v(4);
  IF r(1) == 1 THEN m := r(2); END;
  IF NOT INPUT({m, s, n, u0}, "One-sample z test", {"Mean =", "σ =", "n =", "μ0 ="},
               {"Sample mean", "Population standard deviation σ", "Sample size n", "H0: μ = μ0"}) THEN RETURN; END;
  IF s <= 0 THEN zMsg := "1-Z: need σ > 0"; RETURN; END;
  IF n <> IP(n) OR n < 1 THEN zMsg := "1-Z: need a whole n ≥ 1"; RETURN; END;
  zSV(1, {m, s, n, u0});
  se := s/SQRT(n); c := zZS();
  PCx := (m - u0)/se;
  zTFIN("1-Z  SE = " + zN(se), "n " + zN(n), 2, {}, "z", m - c*se, m + c*se, "μ");
END;

// 2-Z: two-sample z test for μA - μB = Δ0 (σA, σB known).
zZ2(r)
BEGIN
  LOCAL v, ma, mb, sa, sb, na, nb, d0;
  LOCAL se, c, dd;
  v := zTI(2);
  ma := v(1); mb := v(2); sa := v(3); sb := v(4); na := v(5); nb := v(6); d0 := v(7);
  IF r(1) == 1 THEN ma := r(2); END;
  IF NOT INPUT({ma, mb, sa, sb, na, nb, d0}, "Two-sample z test",
               {"Mean A =", "Mean B =", "σ A =", "σ B =", "n A =", "n B =", "Δ0 ="},
               {"Sample mean of A", "Sample mean of B", "Population SD of A", "Population SD of B",
                "Sample size of A", "Sample size of B", "H0: μA - μB = Δ0 (usually 0)"}) THEN RETURN; END;
  IF sa <= 0 OR sb <= 0 THEN zMsg := "2-Z: need σ > 0"; RETURN; END;
  IF na <> IP(na) OR nb <> IP(nb) OR na < 1 OR nb < 1 THEN zMsg := "2-Z: need whole n ≥ 1"; RETURN; END;
  zSV(2, {ma, mb, sa, sb, na, nb, d0});
  se := SQRT(sa^2/na + sb^2/nb); c := zZS(); dd := ma - mb;
  PCx := (dd - d0)/se;
  zTFIN("2-Z  SE = " + zN(se), "diff " + zN(dd), 2, {}, "z", dd - c*se, dd + c*se, "μA-μB");
END;

// 1-P: one-proportion z test for p = p0 from x successes in n.
zP1(r)
BEGIN
  LOCAL v, x, n, p0, ph, se, c;
  v := zTI(3); x := v(1); n := v(2); p0 := v(3);
  IF r(1) == 1 THEN x := r(2); END;
  IF NOT INPUT({x, n, p0}, "One-proportion z test", {"x =", "n =", "p0 ="},
               {"Number of successes x", "Sample size n", "H0: p = p0"}) THEN RETURN; END;
  IF n <> IP(n) OR n < 1 THEN zMsg := "1-P: need a whole n ≥ 1"; RETURN; END;
  IF x < 0 OR x > n THEN zMsg := "1-P: need 0 ≤ x ≤ n"; RETURN; END;
  IF p0 <= 0 OR p0 >= 1 THEN zMsg := "1-P: need 0 < p0 < 1"; RETURN; END;
  zSV(3, {x, n, p0});
  ph := x/n; c := zZS();
  PCx := (ph - p0)/SQRT(p0*(1-p0)/n);
  se := SQRT(ph*(1-ph)/n);
  zTFIN("1-P  x/n = " + zN(ph), "n " + zN(n), 2, {}, "z", ph - c*se, ph + c*se, "p");
END;

// 2-P: two-proportion z test for pA = pB (pooled p in the statistic,
// unpooled SE in the CI).
zP2(r)
BEGIN
  LOCAL v, xa, na, xb, nb, pa, pb, pp;
  LOCAL s0, se, c;
  v := zTI(4); xa := v(1); na := v(2); xb := v(3); nb := v(4);
  IF r(1) == 1 THEN xa := r(2); END;
  IF NOT INPUT({xa, na, xb, nb}, "Two-proportion z test", {"x A =", "n A =", "x B =", "n B ="},
               {"Successes in A", "Sample size of A", "Successes in B", "Sample size of B"}) THEN RETURN; END;
  IF na <> IP(na) OR nb <> IP(nb) OR na < 1 OR nb < 1 THEN zMsg := "2-P: need whole n ≥ 1"; RETURN; END;
  IF xa < 0 OR xa > na OR xb < 0 OR xb > nb THEN zMsg := "2-P: need 0 ≤ x ≤ n"; RETURN; END;
  zSV(4, {xa, na, xb, nb});
  pa := xa/na; pb := xb/nb; pp := (xa + xb)/(na + nb);
  s0 := SQRT(pp*(1-pp)*(1/na + 1/nb));
  IF s0 == 0 THEN zMsg := "2-P: pooled p is 0 or 1"; RETURN; END;
  PCx := (pa - pb)/s0;
  se := SQRT(pa*(1-pa)/na + pb*(1-pb)/nb); c := zZS();
  zTFIN("2-P  pooled p = " + zN(pp), "diff " + zN(pa - pb), 2, {}, "z", pa - pb - c*se, pa - pb + c*se, "pA-pB");
END;

// 1-T (pd = 0) or paired t (pd = 1, on the differences): t test for μ = μ0.
zT1(r, pd)
BEGIN
  LOCAL k, v, m, s, n, u0, se, c;
  k := IFTE(pd, 7, 5);
  v := zTI(k); m := v(1); s := v(2); n := v(3); u0 := v(4);
  IF r(1) == 1 THEN m := r(2); END;
  IF pd THEN
    IF NOT INPUT({m, s, n, u0}, "Paired t test", {"Mean d =", "s d =", "n =", "μd0 ="},
                 {"Mean of the differences d = A - B", "SD of the differences", "Number of pairs n ≥ 2",
                  "H0: μd = μd0 (usually 0)"}) THEN RETURN; END;
  ELSE
    IF NOT INPUT({m, s, n, u0}, "One-sample t test", {"Mean =", "s =", "n =", "μ0 ="},
                 {"Sample mean", "Sample standard deviation s", "Sample size n ≥ 2", "H0: μ = μ0"}) THEN RETURN; END;
  END;
  IF s <= 0 THEN zMsg := "t: need s > 0"; RETURN; END;
  IF n <> IP(n) OR n < 2 THEN zMsg := "t: need a whole n ≥ 2"; RETURN; END;
  zSV(k, {m, s, n, u0});
  se := s/SQRT(n); c := STUDENT_ICDF(n-1, 1-PCa/2);
  PCx := (m - u0)/se;
  zTFIN(IFTE(pd, "Paired", "1-T") + "  SE = " + zN(se), "df " + zN(n-1), 3, {n-1}, "t",
        m - c*se, m + c*se, IFTE(pd, "μd", "μ"));
END;

// 2-T: two-sample t test for μA - μB = Δ0, pooled or Welch.
zT2(r)
BEGIN
  LOCAL v, ma, mb, sa, sb, na, nb, d0;
  LOCAL pl, va, vb, se, df, sp, c, dd;
  v := zTI(6);
  ma := v(1); mb := v(2); sa := v(3); sb := v(4); na := v(5); nb := v(6); d0 := v(7); pl := v(8);
  IF r(1) == 1 THEN ma := r(2); END;
  IF NOT INPUT({ma, mb, sa, sb, na, nb, d0, {pl, {"Pooled", "Welch"}}}, "Two-sample t test",
               {"Mean A =", "Mean B =", "s A =", "s B =", "n A =", "n B =", "Δ0 =", "Var:"},
               {"Sample mean of A", "Sample mean of B", "Sample SD of A", "Sample SD of B",
                "Sample size of A", "Sample size of B", "H0: μA - μB = Δ0 (usually 0)",
                "Pooled: equal variances; Welch: unequal"}) THEN RETURN; END;
  IF sa <= 0 OR sb <= 0 THEN zMsg := "2-T: need s > 0"; RETURN; END;
  IF na <> IP(na) OR nb <> IP(nb) OR na < 2 OR nb < 2 THEN zMsg := "2-T: need whole n ≥ 2"; RETURN; END;
  zSV(6, {ma, mb, sa, sb, na, nb, d0, pl});
  IF pl == 1 THEN
    sp := SQRT(((na-1)*sa^2 + (nb-1)*sb^2)/(na+nb-2));
    se := sp*SQRT(1/na + 1/nb);
    df := na + nb - 2;
  ELSE
    va := sa^2/na; vb := sb^2/nb;
    se := SQRT(va + vb);
    df := (va + vb)^2/(va^2/(na-1) + vb^2/(nb-1));
  END;
  dd := ma - mb;
  PCx := (dd - d0)/se;
  c := STUDENT_ICDF(df, 1-PCa/2);
  zTFIN(IFTE(pl == 1, "Pooled  sp = " + zN(sp), "Welch"), "df " + zN(df), 3, {df}, "t",
        dd - c*se, dd + c*se, "μA-μB");
END;

// χ²-σ: chi-square test for one variance, σ = σ0; CI for σ².
zCV(r)
BEGIN
  LOCAL v, s, n, s0, df, q;
  v := zTI(8); s := v(1); n := v(2); s0 := v(3);
  IF r(1) == 1 THEN s := r(2); END;
  IF NOT INPUT({s, n, s0}, "χ² test for σ", {"s =", "n =", "σ0 ="},
               {"Sample standard deviation s", "Sample size n ≥ 2", "H0: σ = σ0"}) THEN RETURN; END;
  IF s <= 0 OR s0 <= 0 THEN zMsg := "χ²: need s, σ0 > 0"; RETURN; END;
  IF n <> IP(n) OR n < 2 THEN zMsg := "χ²: need a whole n ≥ 2"; RETURN; END;
  zSV(8, {s, n, s0});
  df := n - 1; q := df*s^2;
  PCx := q/s0^2;
  zTFIN("χ² var  s² = " + zN(s^2), "df " + zN(df), 4, {df}, "χ²",
        q/CHISQUARE_ICDF(df, 1-PCa/2), q/CHISQUARE_ICDF(df, PCa/2), "σ²");
END;

// 2-Var F: F test for σA² = σB²; CI for σA²/σB².
zFV(r)
BEGIN
  LOCAL v, sa, sb, na, nb, d1, d2, f;
  v := zTI(9); sa := v(1); sb := v(2); na := v(3); nb := v(4);
  IF r(1) == 1 THEN sa := r(2); END;
  IF NOT INPUT({sa, sb, na, nb}, "Two-variance F test", {"s A =", "s B =", "n A =", "n B ="},
               {"Sample SD of A (numerator)", "Sample SD of B", "Sample size of A", "Sample size of B"}) THEN RETURN; END;
  IF sa <= 0 OR sb <= 0 THEN zMsg := "F: need s > 0"; RETURN; END;
  IF na <> IP(na) OR nb <> IP(nb) OR na < 2 OR nb < 2 THEN zMsg := "F: need whole n ≥ 2"; RETURN; END;
  zSV(9, {sa, sb, na, nb});
  d1 := na - 1; d2 := nb - 1; f := sa^2/sb^2;
  PCx := f;
  zTFIN("2-Var F", "df " + zN(d1) + ", " + zN(d2), 5, {d1, d2}, "F",
        f/FISHER_ICDF(d1, d2, 1-PCa/2), f/FISHER_ICDF(d1, d2, PCa/2), "σA²/σB²");
END;

// Test key: pick a test for the current distribution (list if several),
// then run it. A typed number seeds the dialog's first field.
zTEST()
BEGIN
  LOCAL r, nm, cd, ch;
  r := zGV();
  IF r(1) == -1 THEN RETURN; END;
  CASE
    IF PCd <= 2 THEN
      nm := {"1-Z  one mean (σ known)", "2-Z  two means (σ known)",
             "1-P  one proportion", "2-P  two proportions"};
      cd := {1, 2, 3, 4};
    END;
    IF PCd == 3 THEN
      nm := {"1-T  one mean", "2-T  two means (pooled / Welch)", "Paired t"};
      cd := {5, 6, 7};
    END;
    IF PCd == 4 THEN nm := {}; cd := {8}; END;
    IF PCd == 5 THEN nm := {}; cd := {9}; END;
    DEFAULT RETURN;
  END;
  ch := 1;
  IF SIZE(cd) > 1 THEN
    IF NOT CHOOSE(ch, "Test", nm) THEN RETURN; END;
  END;
  CASE
    IF cd(ch) == 1 THEN zZ1(r); END;
    IF cd(ch) == 2 THEN zZ2(r); END;
    IF cd(ch) == 3 THEN zP1(r); END;
    IF cd(ch) == 4 THEN zP2(r); END;
    IF cd(ch) == 5 THEN zT1(r, 0); END;
    IF cd(ch) == 6 THEN zT2(r); END;
    IF cd(ch) == 7 THEN zT1(r, 1); END;
    IF cd(ch) == 8 THEN zCV(r); END;
    IF cd(ch) == 9 THEN zFV(r); END;
  END;
END;

zDO(a)
BEGIN
  LOCAL r, v, c, s;
  IF a == 0 THEN RETURN; END;
  IF a == -1 THEN zV := 1; zM := 0; zPP := CEILING(PCd/6); RETURN; END;
  IF a > 200 THEN
    PCd := a - 200; zM := 1; zPP := 1;
    zDEFX();
    RETURN;
  END;
  IF a >= 31 AND a <= 34 THEN
    PCt := a - 30;
    IF PCt == 4 AND PCy == PCx THEN PCy := zISF(0.25); END;
    RETURN;
  END;
  CASE
    IF a == 1 THEN zSET(); END;
    IF a == 2 THEN
      r := zARG(PCx);
      IF r(1) == 1 THEN
        PCx := r(2);
        zADD(IFTE(zDS(), "P(X=", "f(") + zN(PCx) + ")", zN(zPDF(PCx)));
      END;
    END;
    IF a == 3 OR a == 4 THEN
      r := zARG(PCx);
      IF r(1) == 1 THEN
        PCx := r(2);
        zADP(IFTE(a == 3, "P(X≤", "P(X≥") + zN(PCx) + ")", IFTE(a == 3, zCDF(PCx), zSF(PCx)));
      END;
    END;
    IF a == 5 THEN
      IF zAB(0) THEN
        PCt := 4;
        zADP("P(" + zN(MIN(PCx,PCy)) + "≤X≤" + zN(MAX(PCx,PCy)) + ")", zPV());
      END;
    END;
    IF a == 6 OR a == 7 THEN
      r := zPRB(IFTE(a == 6, "x for P(X ≤ x) = p", "x for P(X ≥ x) = p"));
      IF r(1) == 1 THEN
        IF a == 6 THEN v := zICDF(r(2)); ELSE v := zISF(r(2)); END;
        PCx := v;
        zADD(zQS() + IFTE(a == 6, "-L(", "-R(") + zN(r(2)) + ")", zN(v));
      END;
    END;
    IF a == 8 THEN
      r := zGV();
      IF r(1) == -1 THEN RETURN; END;
      IF r(1) == 1 THEN
        IF PCt == 4 THEN PCy := r(2); ELSE PCx := r(2); END;
      END;
      IF PCt == 4 THEN
        zADP("P(" + zN(MIN(PCx,PCy)) + "≤X≤" + zN(MAX(PCx,PCy)) + ")", zPV());
      ELSE
        zADP(zPLB(), zPV());
      END;
    END;
    IF a == 9 THEN PCt := PCt MOD 4 + 1; END;
    IF a == 10 THEN
      r := zARG(PCa);
      IF r(1) == 1 THEN
        IF r(2) <= 0 OR r(2) >= 1 THEN zMsg := "Need 0 < α < 1"; ELSE PCa := r(2); zADP("α", PCa); END;
      END;
    END;
    IF a == 11 THEN
      c := zCRIT();
      IF SIZE(c) == 0 THEN zMsg := "Crit: Left, Right or Two"; RETURN; END;
      s := "Crit " + zTN(PCt) + " " + zAT();
      IF zDS() THEN
        v := IFTE(c(1,2) < 0, "X≤", "X≥") + zN(c(1,1));
        IF SIZE(c) == 2 THEN v := v + ", X≥" + zN(c(2,1)); END;
      ELSE
        v := zN(c(1,1));
        IF SIZE(c) == 2 THEN v := v + ", " + zN(c(2,1)); END;
      END;
      PCx := c(SIZE(c), 1);
      zADD(s, v);
    END;
    IF a == 12 THEN zV := 2; END;
    IF a == 13 THEN zH := {}; END;
    IF a == 15 THEN zCI(); END;
    IF a == 16 THEN zTEST(); END;
    IF a == 14 THEN
      IF PCt == 4 THEN
        zAB(1);
      END;
    END;
  END;
END;

// Move x by n plot steps (one bar when discrete).
zMOV(n)
BEGIN
  LOCAL r, st;
  IF zDS() THEN st := 1; ELSE r := zRNG(); st := zNS(r(1), r(2))/10; END;
  PCx := ROUND(ROUND(PCx/st, 0)*st + n*st, 10);
END;

zKEY(k)
BEGIN
  LOCAL i, np;
  i := POS(zKD, k);
  IF i > 0 THEN
    IF DIM(zCmd) < 20 THEN zCmd := zCmd + STRING(i-1, 1); END;
    RETURN;
  END;
  CASE
    IF k == 48 THEN
      IF NOT zHAS(zCmd, ".") AND NOT zHAS(zCmd, zEE()) THEN zCmd := zCmd + "."; END;
    END;
    IF k == 31 THEN
      IF NOT zHAS(zCmd, zEE()) THEN
        IF DIM(zCmd) == 0 OR zCmd == "-" THEN zCmd := zCmd + "1"; END;
        zCmd := zCmd + zEE();
      END;
    END;
    IF k == 27 THEN
      CASE
        IF DIM(zCmd) == 0 THEN zCmd := "-"; END;
        IF RIGHT(zCmd, 1) == zEE() THEN zCmd := zCmd + "-"; END;
        IF RIGHT(zCmd, 2) == zEE() + "-" THEN zCmd := LEFT(zCmd, DIM(zCmd)-1); END;
        IF MID(zCmd, 1, 1) == "-" THEN zCmd := MID(zCmd, 2); END;
        DEFAULT zCmd := "-" + zCmd;
      END;
    END;
    IF k == 19 THEN
      IF DIM(zCmd) > 0 THEN
        zCmd := MID(zCmd, 1, DIM(zCmd)-1);
      ELSE
        IF SIZE(zH) > 1 THEN zH := zH({2, SIZE(zH)}); ELSE zH := {}; END;
      END;
    END;
    IF k == 30 THEN
      IF DIM(zCmd) == 0 AND zV == 1 AND zM == 0 THEN zDO(200 + PCd); RETURN; END;
      IF DIM(zCmd) > 0 THEN
        i := zGV();
        IF i(1) == 1 THEN PCx := i(2); zADD("x", zN(PCx)); END;
      END;
    END;
    IF k == 6  THEN zV := 2; END;
    IF k == 11 THEN zV := 1; END;
    IF k == 1 OR k == 13 THEN
      zV := 1;
      zDO(-1);
    END;
    IF k == 4 THEN
      CASE
        IF DIM(zCmd) > 0 THEN zCmd := ""; END;
        IF zV == 2 THEN zV := 1; END;
        IF zM == 1 THEN zDO(-1); END;
      END;
    END;
    IF k == 7 OR k == 8 THEN
      IF zV == 2 THEN
        zMOV(IFTE(k == 7, -1, 1));
      ELSE
        np := 3;
        IF k == 7 THEN zPP := IFTE(zPP > 1, zPP-1, np); ELSE zPP := IFTE(zPP < np, zPP+1, 1); END;
        // Landing page: the selection jumps to the first row of the new column.
        IF zM == 0 THEN PCd := 6*(zPP-1) + 1; zDEFX(); END;
      END;
    END;
    IF k == 2 OR k == 12 THEN
      IF zV == 2 THEN
        zMOV(IFTE(k == 12, -10, 10));
      ELSE
        IF k == 2 THEN PCd := IFTE(PCd > 1, PCd-1, 14); ELSE PCd := IFTE(PCd < 14, PCd+1, 1); END;
        IF zM == 0 THEN zPP := CEILING(PCd/6); END;
        zDEFX();
      END;
    END;
    IF k == 3 THEN zHELP(); END;
    IF k == 5 THEN zEX := 1; END;
    IF k == 0 THEN zEX := 2; END;
  END;
END;

zTAP(mx, my)
BEGIN
  LOCAL m, pg, rw;
  // Landing page (boxes from zDRAWL): a tap on a name in any box selects it
  // (highlight, and its box becomes the current page); a tap on the name
  // already selected opens it (Enter does too).
  IF zV == 1 AND zM == 0 AND my >= 41 AND my <= 182 AND mx >= 2 AND mx < 316 THEN
    pg := MIN(3, IP((mx-2)/106) + 1);
    IF my < 45 OR my >= 177 THEN RETURN; END;
    rw := IP((my-45)/22) + 1;
    m := 6*(pg-1) + rw;
    IF rw > 6 OR m > 14 THEN RETURN; END;
    IF m == PCd AND pg == zPP THEN zDO(200 + m); ELSE PCd := m; zPP := pg; END;
    RETURN;
  END;
  IF my < 220 THEN RETURN; END;
  m := MIN(6, IP(mx/53) + 1);
  zDO(zCODE(m));
END;

// Repair state left over from an older build or edited from Home.
zFIX()
BEGIN
  IF TYPE(PCp) <> 6 OR SIZE(PCp) <> 14 THEN
    PCp := {{0,1},{},{10},{5},{5,10},{0.2},{0,1},{10,0.5},{0.5},{0.2},{3,0.5},{3},{20,6,5},{20,6,3}};
  END;
  IF PCd < 1 OR PCd > 14 OR PCd <> IP(PCd) THEN PCd := 1; END;
  IF PCt < 1 OR PCt > 4 OR PCt <> IP(PCt) THEN PCt := 3; END;
  IF PCa <= 0 OR PCa >= 1 THEN PCa := 0.05; END;
  IF zPP < 1 OR zPP > 3 THEN zPP := 1; END;
END;

zMAIN()
BEGIN
  LOCAL w, d;
  zEX := 0; d := 1;
  DIMGROB_P(G9, 640, 40);
  zFIX();
  WHILE zEX == 0 DO
    IF d THEN
      IFERR zDRAW(); THEN
        zCmd := "";
        RECT_P(G0, 0, 20, 319, 37, #4F4F4Fh, #4F4F4Fh);
        TEXTOUT_P("Error while drawing", G0, 4, 23, 2, #FF3030h);
        zMENU();
      END;
    END;
    d := 0;
    w := WAIT(-1);
    IF TYPE(w) == 6 THEN
      IF SIZE(w) >= 3 THEN
        IF B→R(w(1)) == 3 THEN
          IFERR zTAP(B→R(w(2)), B→R(w(3))); THEN zMsg := "Error on menu tap"; END;
          d := 1;
        END;
      END;
    ELSE
      // A runtime error would end the app; report it in the strip instead.
      IF w >= 0 THEN
        IFERR zKEY(w); THEN zMsg := "Error on key " + STRING(w, 1); END;
        d := 1;
      END;
    END;
  END;
  IF zEX == 2 THEN STARTVIEW(-4, 1); ELSE STARTVIEW(-1, 1); END;
END;
